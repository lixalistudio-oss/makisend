name: Deploy — Cloudflare Pages

# The project's only deployment path. No local deployment is allowed in any
# form — the rule lives in AGENTS.md (rule 1) and is restated in the README.
#
# Triggers:
#   - push on main            → production deployment
#   - pull_request on main    → CI on every PR, plus a Cloudflare Pages
#                               preview deployment when the PR head branch is
#                               a feature branch (feat/* or redesign/*)
#   - workflow_dispatch       → manual production deployment

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  deployments: write

concurrency:
  group: pages-${{ github.ref }}
  cancel-in-progress: true

jobs:
  build:
    name: Verify and build
    runs-on: ubuntu-latest
    steps:
      - name: Check out the repository
        uses: actions/checkout@v4

      # Must come before setup-node: this is what makes `cache: pnpm` possible.
      # Without `version`, the action reads `packageManager` from package.json.
      - name: Install pnpm
        uses: pnpm/action-setup@v4

      - name: Install Node
        uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Type-check and build
        run: pnpm run build

      # Checks the real dist/: the charter's hard rules (no `color: #C0703A`),
      # deferred hydration of the islands, baseline accessibility, no
      # third-party request. python3 ships with ubuntu-latest runners.
      - name: Check the build against the charter
        run: pnpm run verify:build

      - name: Upload the build
        uses: actions/upload-artifact@v4
        with:
          name: dist
          path: dist/
          retention-days: 7

  deploy:
    name: Deploy to Cloudflare Pages
    needs: build
    if: github.event_name != 'pull_request'
    runs-on: ubuntu-latest
    environment:
      name: production
      url: ${{ steps.deploy.outputs.url }}
    steps:
      - name: Check out the repository
        uses: actions/checkout@v4

      - name: Install pnpm
        uses: pnpm/action-setup@v4

      - name: Install Node
        uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Build
        run: pnpm run build

      - name: Publish to Cloudflare Pages
        id: deploy
        uses: cloudflare/wrangler-action@v3
        with:
          apiToken: ${{ secrets.CLOUDFLARE_API_TOKEN }}
          accountId: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
          command: >-
            pages deploy dist/client
            --project-name=itaja-landing-page
            --branch=${{ github.ref_name }}

  deploy-preview:
    name: Deploy PR preview to Cloudflare Pages
    needs: build
    # Previews only for feature branches of this repository: secrets (and the
    # write token for commenting) are not available on forked PRs.
    if: >-
      github.event_name == 'pull_request' &&
      github.event.pull_request.head.repo.full_name == github.repository &&
      (startsWith(github.head_ref, 'feat/') ||
        startsWith(github.head_ref, 'redesign/'))
    runs-on: ubuntu-latest
    permissions:
      contents: read
      pull-requests: write
    environment:
      name: preview
    steps:
      - name: Download the build
        uses: actions/download-artifact@v4
        with:
          name: dist
          path: dist/

      # --branch makes this a preview deployment (never production): the site
      # is served from <branch>.itaja-landing-page.pages.dev and tracked
      # separately from the production deployment in the Pages dashboard.
      - name: Publish preview to Cloudflare Pages
        id: deploy
        uses: cloudflare/wrangler-action@v3
        with:
          apiToken: ${{ secrets.CLOUDFLARE_API_TOKEN }}
          accountId: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
          command: >-
            pages deploy dist/client
            --project-name=itaja-landing-page
            --branch=${{ github.head_ref }}

      - name: Comment the preview URL on the PR
        uses: actions/github-script@v7
        env:
          DEPLOY_OUTPUT: ${{ steps.deploy.outputs.command-output }}
        with:
          script: |
            const marker = '<!-- cf-pages-preview -->';
            const branch = context.payload.pull_request.head.ref;
            const match = (process.env.DEPLOY_OUTPUT || '').match(/https:\/\/[^\s]+\.pages\.dev/);
            if (!match) {
              core.warning('No pages.dev URL found in the wrangler output');
              return;
            }
            const body = [
              marker,
              '### 🔍 Preview Cloudflare Pages',
              '',
              `**URL : ${match[0]}**`,
              '',
              `Branche \`${branch}\` — se met à jour à chaque push.`,
            ].join('\n');
            const { data: comments } = await github.rest.issues.listComments({
              ...context.repo,
              issue_number: context.issue.number,
            });
            const existing = comments.find(
              (c) => c.user?.type === 'Bot' && c.body?.includes(marker),
            );
            if (existing) {
              await github.rest.issues.updateComment({
                ...context.repo,
                comment_id: existing.id,
                body,
              });
            } else {
              await github.rest.issues.createComment({
                ...context.repo,
                issue_number: context.issue.number,
                body,
              });
            }
